3. Konfigurasi Bobot

3.1 Bobot Komponen MK (Σ=100)

INSERT INTO bobot_komponen_mk (kelas_id, komponen_id, bobot)
SELECT k.id, kp.id, v.bobot
FROM kelas k
CROSS JOIN (VALUES
    ('PART', 10.00),
    ('TUGAS_IND', 20.00),
    ('UTS', 30.00),
    ('UAS', 40.00)
) AS v(kode, bobot)
JOIN komponen_penilaian kp ON kp.kode = v.kode
WHERE k.kode_kelas = 'TI012-A';

4. Query Penting

4.1 Lihat semua CPL

SELECT id, kode, aspek, LEFT(deskripsi, 80) AS deskripsi
FROM cpl ORDER BY kode;

4.2 Lihat matriks CPMK × MK

SELECT c.kode AS cpmk, c.deskripsi, 
       STRING_AGG(m.kode, ', ') AS mk_list
FROM cpmk c
LEFT JOIN cpmk_mk cm ON cm.cpmk_id = c.id
LEFT JOIN mk m ON m.id = cm.mk_id
GROUP BY c.id, c.kode, c.deskripsi
ORDER BY c.kode;

4.3 Nilai akhir MK per mahasiswa

SELECT 
    mhs.nim, mhs.nama,
    mk.kode AS mk_kode, mk.nama AS mk_nama,
    nm.nilai_angka, nm.nilai_huruf,
    nm.bobot_nilai, nm.lulus
FROM nilai_mk nm
JOIN krs_detail kd ON kd.id = nm.krs_detail_id
JOIN krs k ON k.id = kd.krs_id
JOIN mahasiswa mhs ON mhs.id = k.mahasiswa_id
JOIN kelas kl ON kl.id = kd.kelas_id
JOIN mk ON mk.id = kl.mk_id
WHERE mhs.nim = '21.01.0001'
ORDER BY mk.semester, mk.kode;

4.4 Hitung IPK

SELECT 
    mhs.nim, mhs.nama,
    SUM(mk.sks) AS total_sks,
    SUM(nm.bobot_nilai * mk.sks) AS total_mutu,
    ROUND(
        SUM(nm.bobot_nilai * mk.sks) / NULLIF(SUM(mk.sks), 0),
        2
    ) AS ipk
FROM nilai_mk nm
JOIN krs_detail kd ON kd.id = nm.krs_detail_id
JOIN krs k ON k.id = kd.krs_id
JOIN mahasiswa mhs ON mhs.id = k.mahasiswa_id
JOIN kelas kl ON kl.id = kd.kelas_id
JOIN mk ON mk.id = kl.mk_id
WHERE mhs.nim = '21.01.0001'
  AND k.status = 'disetujui'
GROUP BY mhs.id, mhs.nim, mhs.nama;

4.5 Ketercapaian CPL per mahasiswa

SELECT 
    cpl.kode AS cpl_kode,
    kc.skor_maks,
    kc.total_skor,
    kc.nilai_rata_rata,
    kc.status
FROM ketercapaian_cpl kc
JOIN cpl ON cpl.id = kc.cpl_id
JOIN mahasiswa mhs ON mhs.id = kc.mahasiswa_id
WHERE mhs.nim = '21.01.0001'
ORDER BY cpl.kode;

5. Update & Delete

5.1 Update nilai komponen

UPDATE nilai_komponen
SET nilai = 90
WHERE krs_detail_id = 1 AND komponen_id = 6;

5.2 Setujui KRS

UPDATE krs
SET status = 'disetujui',
    approved_by = 1,
    approved_at = NOW()
WHERE id = 1;

5.3 Hapus kelas (jika belum ada peserta)

DELETE FROM kelas
WHERE id = 1
  AND NOT EXISTS (
      SELECT 1 FROM krs_detail WHERE kelas_id = 1
  );
